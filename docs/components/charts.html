  title: 'Monthly Sales',

  xAxis: {
    key: 'month',
    type: 'time',
    title: 'Month',
    timeGrain: 'month',

  yAxis: {
    title: 'Amount ($)',
    echartOptions: {
      min: 0,
      max: 800,
    },
  },
  series: [{ name: 'sales', type: 'bar' }],

const stackedConfig: AxisChartConfig = {
  data: salesData,

    title: 'Month',

    { name: 'growth_rate', type: 'line', axis: 'y2' },
